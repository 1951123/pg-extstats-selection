\begin{abstract}
TODO: one-paragraph abstract. Draft: we observe that PostgreSQL
\texttt{CREATE STATISTICS} selection is not just ``which column combinations''
but a joint decision of \emph{which} (table, columns) and \emph{how much}
capacity (each statistic's \texttt{statistics\_target}). We re-frame it as
\emph{capacity-aware selection}: under a storage budget, is the right choice a
smaller set of statistics, or the same statistics stored at a lower capacity?
Across single-table workloads (statistics\_CEB and the wide 69-column CENSUS
benchmark) we find a strikingly sparse structure: one well-chosen multivariate
statistic captures nearly all attainable improvement per query, making the
objective exactly linear and the resulting budgeted MILP solvable in seconds at
full-workload scale. End-to-end validation on PostgreSQL 16 shows individual
candidate measurements are predictive, but reveals that co-installed
overlapping statistics cause planner interference --- motivating global
disjointness or post-selection validation. Join-heavy workloads act as a
negative control.
\end{abstract}

\subsection{Multi-select vs.\ the Sparse Special Case}
\label{sec:sparse}

% General multi-select needs a multiplicative approximation in log space; the
% approximation breaks for column-overlapping stats (joint interference).
% Sparse special case (this paper's model): at most one statistic per query
% (per_query_cap=1) makes the objective EXACTLY linear:
% \min \sum_i ( e_i^0 - \sum_s (e_i^0 - e_is) x_is ),  s.t. sum_s x_is <= 1.

\begin{proposition}[Exact linearity under per-query unit cap]
\label{prop:linear}
If each query selects at most one statistic ($\sum_s x_{is} \le 1$), then the
per-query q-error is exactly
\[
  e_i\big(T_i\big)\;=\;e_i^0\;-\;\sum_s \big(e_i^0-e_{is}\big)\,x_{is},
\]
with no interaction term, so the workload objective is linear in the selection
variables.
\end{proposition}

% statistics, hence no joint/interaction term in q-error.

\subsection{Option-A Ablation: Global Disjointness Restores Exact Predictability}
\label{sec:p1}

% We implemented the Option-A model: a ``global\_disjoint'' constraint in the
% MILP that forbids co-installing any two physical stats whose column sets
% overlap (y\_a + y\_b \le 1 when columns(a) intersect columns(b)), and measured
% predicted vs E2E for both models (stats\_CEB single-table top-10; docs §5.9):
%
% | budget | model          | stats | predicted | E2E  | ratio |
% |--------|----------------|------:|----------:|-----:|------:|
% | 2 MB   | overlap allowed| 5     | 1.021     | 1.320| 1.294 |
% | 2 MB   | global-disjoint| 2     | 2.004     | 2.004| 1.000 |
% | 100 KB | overlap allowed| 7     | 1.632     | 1.933| 1.184 |
% | 100 KB | global-disjoint| 2     | 2.004     | 2.004| 1.000 |
%
% Finding: the global-disjoint solution is EXACTLY predictable (ratio 1.000 at
% both budgets), because the deployed set is pairwise column-disjoint so the
% planner cannot cross-talk (the §3 independence assumption holds structurally).
% The cost is a quantified nominal-quality loss (predicted 2.00 vs 1.02):
% disjointness forces the solver to give up overlapping-but-high-value dominant
% candidates.
%
% This sharply isolates two facts: (i) per-candidate measurement is sound
% (disjoint ratios are 1.000); (ii) the overlap model's residual error is
% entirely planner interference among co-installed overlapping stats
% (ratio 1.18-1.29). Option A therefore gives a *trustworthy* deployment at a
% quantified quality cost.
